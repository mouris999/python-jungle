import threading
import time

def worker(n):
    print(f'Thread {n} starting')
    time.sleep(0.1)
    print(f'Thread {n} done')

t1 = threading.Thread(target=worker, args=(1,))
t2 = threading.Thread(target=worker, args=(2,))
t1.start()
t2.start()
t1.join()
t2.join()
print('Both threads completed!')
