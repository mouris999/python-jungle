import sqlite3
conn = sqlite3.connect('jungle.db')
cursor = conn.cursor()
cursor.execute('CREATE TABLE IF NOT EXISTS creatures(name TEXT, age INTEGER)')
cursor.execute("INSERT INTO creatures VALUES ('Monkey', 5)")
cursor.execute("INSERT INTO creatures VALUES ('Parrot', 2)")
cursor.execute("INSERT INTO creatures VALUES ('Snake', 10)")
cursor.execute('SELECT * FROM creatures')
print('All:', cursor.fetchall())
cursor.execute('UPDATE creatures SET age=3 WHERE name="Parrot"')
cursor.execute('DELETE FROM creatures WHERE name="Snake"')
cursor.execute('SELECT * FROM creatures')
print('After updates:', cursor.fetchall())
conn.close()
