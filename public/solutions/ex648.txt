class Singleton(type):
    _instances = {}
    def __call__(cls, *a, **kw):
        if cls not in cls._instances:
            cls._instances[cls] = super().__call__(*a, **kw)
        return cls._instances[cls]
class JungleDB(metaclass=Singleton):
    def __init__(self):
        print("DB Created!")
db1 = JungleDB()
db2 = JungleDB()
print(db1 is db2)
